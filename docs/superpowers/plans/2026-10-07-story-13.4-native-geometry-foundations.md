## Global Constraints

- Keep the existing iPhone-first chapter reader, local progress model, and View Original Page escape hatch.
- Keep one existing image pipeline with maximum three concurrent fetch/decode operations. Do not introduce eager whole-chapter loading, additional pipelines, zoom persistence, or a schema migration.
- Retain the existing bound of visible page containers plus two.
- Let V be the count of page rectangles with positive-area intersection with the actual content viewport, after accounting for scale and insets. Merely touching an edge is not visible.
- For V greater than zero, materialize all visible pages and at most two non-visible neighbors, one before and one after where available.
- For V equal to zero, materialize the nearest page plus at most one adjacent page, never more than two. Ties choose the lower nearest-page index. Choose the adjacent page toward the viewport center relative to the nearest page's center; an exact center tie chooses the following page.
- No normalized anchor is added to persistent storage by this scope.

## Scope and explicit geometry conventions

This implements the page-rectangle/normalized-anchor and container-selection portions of delivery Stage 1. Constructing fitted layouts from decoded metadata, focal translation, pan bounds, transaction queues, side-effect fencing, and native measurement are separate follow-on outcomes. Do not claim those requirements are complete from this slice.

Input rectangles are already in one unscaled logical chapter coordinate space. The future adapter converts the usable inset-adjusted viewport into that same space before calling the selector. Array position is the ordered page index for one fixed-order chapter session; a new chapter or reordered page payload must use a new session ID. Snapshot revision is caller-supplied; projection deliberately permits a newer snapshot of the same session to reconstruct the same normalized point. A future transaction coordinator, not this immutable value, decides whether a revision is current.

Rectangles must have positive, representable finite extents, be vertically ordered and non-overlapping; gaps and variable horizontal positions/widths are allowed. Reject invalid geometry with a failable initializer, not fabricated dimensions. Anchor capture in a gap or outside all pages returns nil. Anchor capture exactly on a shared page border chooses the lower page index and includes page edges; this does not change positive-area visibility selection. A foreign session, unavailable index, or invalid normalized coordinate returns nil during projection.

A valid empty chapter yields an empty selection. An invalid viewport or overflowing derived distance yields nil, distinguishing an unavailable measurement from a valid empty selection. The adapter must not treat nil as native acceptance or permission to read further. These functions calculate desired indexes only: actual container creation, reuse and eviction remain native responsibilities.

At zero-visible chapter edges, use the available opposite adjacent page if the preferred direction has no page; for a one-page chapter use only that page. This makes the approved “at most one adjacent” rule executable without manufacturing containers or exceeding the two-container cap. Visible-page selection does not fill unused neighbor slots or fill horizontal visibility gaps with extra pages.

```swift
import CoreGraphics
import Foundation

public struct ReaderSemanticAnchor: Equatable, Sendable {
    public let sessionID: UUID
    public let pageIndex: Int
    public let unitPoint: CGPoint

    public init(sessionID: UUID, pageIndex: Int, unitPoint: CGPoint) {
        self.sessionID = sessionID
        self.pageIndex = pageIndex
        self.unitPoint = unitPoint
    }
}

/// Logical geometry only; revision identity does not grant progress permission.
public struct ReaderChapterGeometry: Equatable, Sendable {
    public let sessionID: UUID
    public let revision: UInt64
    public let pageFrames: [CGRect]

    public init?(sessionID: UUID, revision: UInt64, pageFrames: [CGRect]) {
        var previousBottom: CGFloat?
        for frame in pageFrames {
            guard Self.isUsable(frame) else { return nil }
            if let previousBottom, frame.minY < previousBottom { return nil }
            previousBottom = frame.maxY
        }
        self.sessionID = sessionID
        self.revision = revision
        self.pageFrames = pageFrames
    }

    public func anchor(at point: CGPoint) -> ReaderSemanticAnchor? {
        guard point.x.isFinite, point.y.isFinite,
              let index = pageFrames.firstIndex(where: { frame in
                  point.x >= frame.minX && point.x <= frame.maxX &&
                  point.y >= frame.minY && point.y <= frame.maxY
              }) else { return nil }
        let frame = pageFrames[index]
        let x = (point.x - frame.minX) / frame.width
        let y = (point.y - frame.minY) / frame.height
        guard x.isFinite, y.isFinite else { return nil }
        return ReaderSemanticAnchor(sessionID: sessionID, pageIndex: index,
            unitPoint: CGPoint(x: min(1, max(0, x)), y: min(1, max(0, y))))
    }

    public func point(for anchor: ReaderSemanticAnchor) -> CGPoint? {
        let unit = anchor.unitPoint
        guard anchor.sessionID == sessionID, pageFrames.indices.contains(anchor.pageIndex),
              unit.x.isFinite, unit.y.isFinite,
              (0...1).contains(unit.x), (0...1).contains(unit.y) else { return nil }
        let frame = pageFrames[anchor.pageIndex]
        let point = CGPoint(x: frame.minX + frame.width * unit.x,
                            y: frame.minY + frame.height * unit.y)
        guard point.x.isFinite, point.y.isFinite else { return nil }
        return point
    }

    static func isUsable(_ frame: CGRect) -> Bool {
        !frame.isNull && !frame.isInfinite &&
        frame.origin.x.isFinite && frame.origin.y.isFinite &&
        frame.size.width.isFinite && frame.size.height.isFinite &&
        frame.size.width > 0 && frame.size.height > 0 &&
        frame.minX.isFinite && frame.maxX.isFinite &&
        frame.minY.isFinite && frame.maxY.isFinite &&
        frame.midX.isFinite && frame.midY.isFinite &&
        frame.maxX > frame.minX && frame.maxY > frame.minY
    }
}
```

```swift
import CoreGraphics
import Foundation

/// Desired indexes only. Does not create containers or grant progress permission.
public enum ReaderPageMaterializationPolicy {
    public static func indexes(in geometry: ReaderChapterGeometry, viewport: CGRect) -> [Int]? {
        guard ReaderChapterGeometry.isUsable(viewport) else { return nil }
        let frames = geometry.pageFrames
        guard !frames.isEmpty else { return [] }
        let visible = frames.indices.filter { index in
            let frame = frames[index]
            return min(frame.maxX, viewport.maxX) > max(frame.minX, viewport.minX) &&
                   min(frame.maxY, viewport.maxY) > max(frame.minY, viewport.minY)
        }
        if let first = visible.first, let last = visible.last {
            var selected = visible
            if first > 0 { selected.append(first - 1) }
            if last + 1 < frames.count { selected.append(last + 1) }
            return Array(Set(selected)).sorted()
        }

        var nearest = 0
        var nearestDistance = CGFloat.infinity
        for index in frames.indices {
            let frame = frames[index]
            let distance: CGFloat
            if frame.maxY < viewport.minY {
                distance = viewport.minY - frame.maxY
            } else if frame.minY > viewport.maxY {
                distance = frame.minY - viewport.maxY
            } else {
                distance = 0
            }
            guard distance.isFinite else { return nil }
            if distance < nearestDistance {
                nearest = index
                nearestDistance = distance
            }
        }
        let direction = viewport.midY >= frames[nearest].midY ? 1 : -1
        let preferred = nearest + direction
        let opposite = nearest - direction
        let neighbor = frames.indices.contains(preferred) ? preferred : opposite
        var selected = [nearest]
        if frames.indices.contains(neighbor) { selected.append(neighbor) }
        return selected.sorted()
    }
}
```
